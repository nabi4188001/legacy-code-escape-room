PLAN: The interest run rounds through double, so exact half cents land on the wrong side. I will calculate the interest with BigDecimal and HALF_UP.

=== FILE: rusty-bank/src/main/java/com/rustybank/legacy/InterestCalculator.java ===
package com.rustybank.legacy;

import java.math.BigDecimal;
import java.math.RoundingMode;

public class InterestCalculator {

    public double monthlyInterest(double balance, double annualRatePercent) {
        return BigDecimal.valueOf(balance)
                .multiply(BigDecimal.valueOf(annualRatePercent))
                .divide(BigDecimal.valueOf(1200), 2, RoundingMode.HALF_UP)
                .doubleValue();
    }
}
